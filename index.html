1. Composite Index Creation (DDL)

                
Composite index speeds up multi-criteria filtering across 10,000+ student records.

                

                    
-- Speed up CGPA, Branch & Backlog filtering
CREATE INDEX idx_student_eligibility 
ON Students (branch, cgpa DESC, active_backlogs ASC);

-- Speed up rank window query performance
CREATE INDEX idx_test_scores 
ON AcademicRecords (coding_score DESC, cgpa DESC);

                    
                

            


            
            

                
2. Company Criteria Filtering Query (DML)

                

                    
SELECT 
    s.student_id,
    s.full_name,
    s.branch,
    a.cgpa,
    a.active_backlogs,
    a.coding_score
FROM Students s
JOIN AcademicRecords a ON s.student_id = a.student_id
WHERE a.cgpa >= 7.5
  AND a.active_backlogs <= 0
  AND s.branch IN ('CSE', 'IT', 'ECE')
ORDER BY a.cgpa DESC;

                    
                

            


            
            

                
3. Candidate Ranking with Window Functions

                

                    
SELECT 
    s.full_name,
    s.branch,
    a.coding_score,
    a.cgpa,
    DENSE_RANK() OVER (
        PARTITION BY s.branch 
        ORDER BY a.coding_score DESC, a.cgpa DESC
    ) AS branch_dense_rank,
    ROW_NUMBER() OVER (
        PARTITION BY s.branch 
        ORDER BY a.coding_score DESC, a.cgpa DESC
    ) AS branch_row_num,
    RANK() OVER (
        PARTITION BY s.branch 
        ORDER BY a.coding_score DESC
    ) AS branch_rank
FROM Students s
JOIN AcademicRecords a ON s.student_id = a.student_id;

                    
                

            


            
            

                
4. Branch Wise Placement Summary (CTE)

                

                    
WITH PlacementSummary AS (
    SELECT 
        s.branch,
        COUNT(s.student_id) AS total_students,
        SUM(CASE WHEN s.placement_status = 'Placed' THEN 1 ELSE 0 END) AS placed_students,
        ROUND(AVG(a.cgpa), 2) AS avg_cgpa
    FROM Students s
    JOIN AcademicRecords a ON s.student_id = a.student_id
    GROUP BY s.branch
)
SELECT 
    branch,
    total_students,
    placed_students,
    ROUND((placed_students * 100.0 / total_students), 2) AS placement_percentage,
    avg_cgpa
FROM PlacementSummary
ORDER BY placement_percentage DESC;
